1//! Demonstrates `BMFont` fonts whose glyphs are packed into the four channels of
2//! their texture.
3//!
4//! Both fonts are generated by `AngelCode` `BMFont` from `FiraSans-Bold.ttf`; the
5//! configurations are `assets/bmf_packed.bmfc` and
6//! `assets/bmf_packed_outline.bmfc`.
7//! - `bmf_packed.txt.fnt` stores plain glyphs in each channel.
8//! - `bmf_packed_outline.txt.fnt` stores glyph and outline together in each
9//!   channel. It draws as a glyph in the text's color with a black outline, or
10//!   with the outline in its own color when the text has an outline color.
11//!
12//! Each font is drawn as sprite text and as pre-rendered text.
13
14#![expect(
15    clippy::mod_module_files,
16    reason = "if present as common.rs, cargo thinks it's an example binary"
17)]
18
19use bevy::color::palettes::tailwind;
20use bevy::prelude::*;
21use bevy_asset_loader::prelude::AssetCollectionApp as _;
22use bevy_image_font::atlas_sprites::ImageFontSpriteText;
23use bevy_image_font::rendered::ImageFontPreRenderedText;
24use bevy_image_font::{ImageFont, ImageFontPlugin, ImageFontText};
25
26use crate::common::{DemoAssets, PANGRAMS};
27
28mod common;
29
30fn main() {
31    App::new()
32        .add_plugins((
33            DefaultPlugins.set(ImagePlugin::default_nearest()),
34            ImageFontPlugin,
35        ))
36        .init_collection::<DemoAssets>()
37        .add_systems(Startup, setup)
38        .insert_resource(ClearColor(Color::srgb(0.5, 0.5, 0.5)))
39        .run();
40}
41
42/// How a line of text is drawn.
43struct Line<'assets> {
44    /// The font the line is drawn with.
45    font: &'assets Handle<ImageFont>,
46    /// The color of the glyphs.
47    color: Color,
48    /// The color of the outline, for the font that has one.
49    outline_color: Option<Color>,
50    /// Whether the line is pre-rendered text instead of sprite text.
51    pre_rendered: bool,
52}
53
54/// Spawns sprite text and pre-rendered text for each packed font, one line
55/// each, and for the outlined font once more with a colored outline.
56fn setup(mut commands: Commands, assets: Res<DemoAssets>) {
57    commands.spawn(Camera2d);
58
59    let text = PANGRAMS[0];
60    let outline_color = Some(tailwind::RED_700.into());
61    let lines = [
62        Line {
63            font: &assets.bmf_packed,
64            color: tailwind::YELLOW_300.into(),
65            outline_color: None,
66            pre_rendered: false,
67        },
68        Line {
69            font: &assets.bmf_packed,
70            color: tailwind::SKY_300.into(),
71            outline_color: None,
72            pre_rendered: true,
73        },
74        Line {
75            font: &assets.bmf_packed_outline,
76            color: tailwind::YELLOW_300.into(),
77            outline_color: None,
78            pre_rendered: false,
79        },
80        Line {
81            font: &assets.bmf_packed_outline,
82            color: tailwind::SKY_300.into(),
83            outline_color: None,
84            pre_rendered: true,
85        },
86        Line {
87            font: &assets.bmf_packed_outline,
88            color: tailwind::YELLOW_300.into(),
89            outline_color,
90            pre_rendered: false,
91        },
92        Line {
93            font: &assets.bmf_packed_outline,
94            color: tailwind::SKY_300.into(),
95            outline_color,
96            pre_rendered: true,
97        },
98    ];
99
100    for (line_no, line) in lines.iter().enumerate() {
101        let font_text = ImageFontText::default().text(text).font(line.font.clone());
102        #[expect(
103            clippy::cast_precision_loss,
104            reason = "the magnitude of the numbers we're working on here are too small to lose \
105            anything"
106        )]
107        let transform = Transform::from_translation(Vec3::new(0., 150. - line_no as f32 * 60., 0.));
108
109        if line.pre_rendered {
110            commands.spawn((
111                ImageFontPreRenderedText::default()
112                    .color(line.color)
113                    .outline_color(line.outline_color),
114                font_text,
115                transform,
116            ));
117        } else {
118            commands.spawn((
119                ImageFontSpriteText::default()
120                    .color(line.color)
121                    .outline_color(line.outline_color),
122                font_text,
123                transform,
124            ));
125        }
126    }
127}
