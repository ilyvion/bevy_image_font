1//! Demonstrates multi-line text: each `\n` in the text starts a new line at
2//! the left edge, one line height below the previous one.
3//!
4//! The same text is drawn three times:
5//! - As sprite text (`ImageFontSpriteText`) with the example image font.
6//! - As pre-rendered text (`ImageFontPreRenderedText`) with the example image
7//!   font.
8//! - As sprite text with a `BMFont` font, whose lines are spaced by the font's
9//!   `lineHeight`. Its heading uses Windows (`\r\n`) line endings, which break
10//!   lines the same way as `\n`.
11
12#![expect(
13    clippy::mod_module_files,
14    reason = "if present as common.rs, cargo thinks it's an example binary"
15)]
16
17use bevy::prelude::*;
18use bevy::sprite::Anchor;
19use bevy_asset_loader::prelude::AssetCollectionApp as _;
20use bevy_image_font::atlas_sprites::ImageFontSpriteText;
21use bevy_image_font::rendered::ImageFontPreRenderedText;
22use bevy_image_font::{ImageFontPlugin, ImageFontText};
23
24use crate::common::{DemoAssets, RAINBOW};
25
26mod common;
27
28fn main() {
29    App::new()
30        .add_plugins((
31            DefaultPlugins.set(ImagePlugin::default_nearest()),
32            ImageFontPlugin,
33        ))
34        .init_collection::<DemoAssets>()
35        .add_systems(Startup, setup)
36        .insert_resource(ClearColor(Color::srgb(0.2, 0.2, 0.2)))
37        .run();
38}
39
40/// The text shown below each heading, split over several lines.
41const TEXT: &str = "Sphinx of\nblack quartz,\njudge my vow!";
42
43/// Spawns the three multi-line texts side by side, each anchored at its
44/// top-left corner.
45fn setup(mut commands: Commands, assets: Res<DemoAssets>) {
46    commands.spawn(Camera2d);
47
48    commands.spawn((
49        ImageFontSpriteText::default()
50            .anchor(Anchor::TOP_LEFT)
51            .color(RAINBOW[0]),
52        ImageFontText::default()
53            .text(format!("Atlased sprites\n\n{TEXT}"))
54            .font(assets.example.clone())
55            .font_height(24.0),
56        Transform::from_translation(Vec3::new(-600., 150., 0.)),
57    ));
58
59    // Pre-rendered text is a single sprite holding every line, so the sprite's
60    // anchor positions the whole block.
61    commands.spawn((
62        ImageFontPreRenderedText::default().color(RAINBOW[3]),
63        ImageFontText::default()
64            .text(format!("Rendered sprite\n\n{TEXT}"))
65            .font(assets.example.clone())
66            .font_height(24.0),
67        Anchor::TOP_LEFT,
68        Transform::from_translation(Vec3::new(-200., 150., 0.)),
69    ));
70
71    commands.spawn((
72        ImageFontSpriteText::default()
73            .anchor(Anchor::TOP_LEFT)
74            .color(RAINBOW[4]),
75        ImageFontText::default()
76            .text(format!("BMFont\r\n\r\n{TEXT}"))
77            .font(assets.bmf_txt.clone()),
78        Transform::from_translation(Vec3::new(200., 150., 0.)),
79    ));
80}
