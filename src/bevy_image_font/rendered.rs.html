69/// A component for displaying in-world text that has been pre-rendered using an
70/// image font.
71///
72/// This component requires an `ImageFontText` component for determining its
73/// font and text. It renders its text into an image and sets it as the texture
74/// on its `Sprite` component. The `Sprite`'s color tints the whole image,
75/// including any outline drawn in `outline_color`.
76#[derive(Component, Debug, Clone, Reflect, Setters)]
77#[setters(into)]
78#[require(ImageFontText, Sprite)]
79#[non_exhaustive]
80pub struct ImageFontPreRenderedText {
81    /// The color the glyphs are drawn in, as a tint multiplied with the font
82    /// texture.
83    ///
84    /// The default value is white, which draws the glyphs as they are in the
85    /// font texture.
86    pub color: Color,
87
88    /// The color of the outline, for fonts whose textures hold an outline
89    /// around their glyphs (see [`ImageFont::outline_layers`]).
90    ///
91    /// When `None`, the outline keeps the color it has in the font texture,
92    /// tinted by `color` like the rest of the glyph. When set, every outline is
93    /// drawn in this color first, then the inside of every glyph in `color` on
94    /// top. Fonts without an outline ignore this setting.
95    ///
96    /// The default value is `None`.
97    pub outline_color: Option<Color>,
98}
99
100impl Default for ImageFontPreRenderedText {
101    fn default() -> Self {
102        Self {
103            color: Color::WHITE,
104            outline_color: None,
105        }
106    }
107}

244/// Renders text into images and assigns the resulting image handles to their
245/// holders.
246///
247/// This function is designed to work with any type that implements
248/// [`ImageHandleHolder`], allowing it to be used with multiple components, such
249/// as sprites and UI elements.
250///
251/// # Parameters
252/// - `font_text_to_image_iter`: An iterator over each [`ImageFontText`], the
253///   colors to draw it in, and a mutable reference to the object implementing
254///   [`ImageHandleHolder`] that receives the rendered image.
255/// - `image_fonts`: A reference to the font assets used for rendering.
256/// - `images`: A mutable reference to the collection of image assets. This is
257///   used to store the newly rendered images.
258/// - `layouts`: A reference to the collection of texture atlas assets.
259///
260/// The function iterates over the provided items, renders the text for each
261/// [`ImageFontText`], and updates the corresponding [`ImageHandleHolder`] with
262/// the handle to the newly created image.
263///
264/// # Errors
265/// If text rendering fails for an item (e.g., due to missing font assets
266/// or invalid texture layouts), an error message is logged, and the
267/// corresponding holder is not updated.
268fn render_text_to_image_holder<'borrow>(
269    font_text_to_image_iter: impl Iterator<
270        Item = (
271            &'borrow ImageFontText,
272            TextColors,
273            &'borrow mut (impl ImageHandleHolder + 'borrow),
274        ),
275    >,
276    image_fonts: &Assets<ImageFont>,
277    images: &mut Assets<Image>,
278    layouts: &Assets<TextureAtlasLayout>,
279) {
280    for (image_font_text, colors, image_handle_holder) in font_text_to_image_iter {
281        bevy_log::debug!("Rendering [{}]", image_font_text.text);
282        match render_text_to_image(image_font_text, colors, image_fonts, images, layouts) {
283            Ok(image) => {
284                image_handle_holder.set_image_handle(images.add(image));
285            }
286            Err(error) => {
287                bevy_log::error!(
288                    "Error when rendering image font text {:?}: {}",
289                    image_font_text,
290                    error
291                );
292            }
293        }
294    }
295}

297/// Renders the text from an [`ImageFontText`] into a single image.
298///
299/// This function takes a reference to an [`ImageFontText`] component and
300/// generates an image representation of the text. It applies font-specific
301/// filtering, determines appropriate character placements, and composites the
302/// final output.
303///
304/// # Parameters
305/// - `image_font_text`: The text to render, along with its associated font.
306/// - `colors`: The colors to draw the text in.
307/// - `image_fonts`: The collection of available font assets.
308/// - `images`: The collection of image assets used to retrieve font textures.
309/// - `layouts`: The texture atlas layouts defining character positioning.
310///
311/// # Returns
312/// A [`Result<Image, ImageFontRenderError>`] containing the generated image if
313/// successful, or an error if rendering fails (e.g., due to missing assets).
314///
315/// # Behavior
316/// - If the text is empty, a **1x1 transparent image** is returned to avoid
317///   invalid texture sizes.
318/// - The function leverages [`RenderContext`] to compute character positions
319///   and generate the image.
320fn render_text_to_image(
321    image_font_text: &ImageFontText,
322    colors: TextColors,
323    image_fonts: &Assets<ImageFont>,
324    images: &Assets<Image>,
325    layouts: &Assets<TextureAtlasLayout>,
326) -> Result<Image, ImageFontRenderError> {
327    let (image_font, textures, render_context) =
328        prepare_context(image_font_text, image_fonts, images, layouts)?;
329
330    if render_context.is_empty() {
331        // Can't make a 0x0 image, so make a 1x1 transparent black pixel
332        return Ok(Image::new(
333            Extent3d {
334                width: 1,
335                height: 1,
336                depth_or_array_layers: 1,
337            },
338            TextureDimension::D2,
339            vec![0, 0, 0, 0],
340            TextureFormat::Rgba8UnormSrgb,
341            RenderAssetUsages::RENDER_WORLD,
342        ));
343    }
344
345    let (placements, size) = layout_glyphs(&render_context);
346    let mut output_image = image::RgbaImage::new(size.x, size.y);
347    let pages = textures
348        .iter()
349        .enumerate()
350        .map(|(page_index, texture)| {
351            let texture = image_buffer(texture).ok_or(ImageFontRenderError::UnknownError)?;
352            let outline_layers = colors
353                .outline_color
354                .and_then(|_| image_font.page_outline_layers(page_index))
355                .map(|layers| {
356                    let layer = |handle| {
357                        let image = images
358                            .get(handle)
359                            .ok_or(ImageFontRenderError::MissingTextureAsset)?;
360                        image_buffer(image).ok_or(ImageFontRenderError::UnknownError)
361                    };
362                    Ok::<_, ImageFontRenderError>(OutlineLayerBuffers {
363                        outline: layer(&layers.outline)?,
364                        glyph: layer(&layers.glyph)?,
365                    })
366                })
367                .transpose()?;
368            Ok(PageBuffers {
369                texture,
370                outline_layers,
371            })
372        })
373        .collect::<Result<Vec<_>, ImageFontRenderError>>()?;
374
375    render_glyphs_to_image(
376        image_font,
377        &render_context,
378        &placements,
379        &mut output_image,
380        &pages,
381        colors,
382    );
383
384    #[expect(
385        clippy::cast_possible_truncation,
386        clippy::cast_sign_loss,
387        clippy::cast_precision_loss,
388        reason = "the magnitude of the numbers we're working on here are too small to lose anything"
389    )]
390    if let Some(font_height) = image_font_text.font_height {
391        let line_height = render_context.max_height() as f32;
392        let scale_dimension =
393            |dimension: u32| (dimension as f32 * font_height / line_height) as u32;
394
395        output_image = imageops::resize(
396            &output_image,
397            scale_dimension(size.x),
398            scale_dimension(size.y),
399            FilterType::Nearest,
400        );
401    }
402
403    let mut bevy_image = Image::new(
404        Extent3d {
405            width: output_image.width(),
406            height: output_image.height(),
407            depth_or_array_layers: 1,
408        },
409        TextureDimension::D2,
410        output_image.into_vec(),
411        TextureFormat::Rgba8UnormSrgb,
412        RenderAssetUsages::RENDER_WORLD,
413    );
414    bevy_image.sampler = ImageSampler::nearest();
415
416    Ok(bevy_image)
417}

419/// The position of a glyph's top-left corner in the rendered text image, at
420/// the font's native size.
421#[derive(Debug, Clone, Copy, PartialEq, Eq)]
422struct GlyphPlacement {
423    /// The character whose glyph is drawn.
424    character: char,
425    /// The glyph's position, in pixels from the image's top-left corner.
426    position: I64Vec2,
427}
428
429/// Positions every glyph of the filtered text at the font's native size.
430///
431/// Each line starts at the left edge, one line height below the line before
432/// it. Each glyph is drawn at the pen position plus its offsets, and the pen
433/// then moves by the character's advance. The image covers every line (the
434/// widest line's sum of advances by the line count times the line height) as
435/// well as any glyph that extends past them.
436///
437/// # Parameters
438/// - `render_context`: The context providing the filtered text and font
439///   metrics.
440///
441/// # Returns
442/// The glyph placements, and the size of the image that holds them.
443#[expect(
444    clippy::cast_possible_truncation,
445    clippy::cast_possible_wrap,
446    reason = "the magnitude of the numbers we're working on here are too small to lose anything"
447)]
448fn layout_glyphs(render_context: &RenderContext<'_>) -> (Vec<GlyphPlacement>, UVec2) {
449    let mut placements = Vec::new();
450    let line_height = i64::from(render_context.max_height());
451    let mut min = I64Vec2::ZERO;
452    let mut max = I64Vec2::new(0, i64::from(render_context.text_height()));
453
454    for (line_index, line) in render_context.lines().iter().enumerate() {
455        let line_top = line_index as i64 * line_height;
456        let mut pen = 0.0;
457
458        let mut characters = line.filtered_chars().peekable();
459        while let Some(character) = characters.next() {
460            let offsets = render_context.character_offsets(character);
461            let position = I64Vec2::new(
462                (pen + offsets.x).round() as i64,
463                line_top + (-offsets.y).round() as i64,
464            );
465            let size = render_context.glyph_rect(character).size().as_i64vec2();
466
467            min = min.min(position);
468            max = max.max(position + size);
469            placements.push(GlyphPlacement {
470                character,
471                position,
472            });
473
474            pen += render_context.character_advance(character, characters.peek().copied());
475        }
476        max.x = max.x.max(pen.round() as i64);
477    }
478
479    for placement in &mut placements {
480        placement.position -= min;
481    }
482
483    let size = (max - min).as_uvec2();
484
485    (placements, size)
486}

488/// A font image's pixel data, borrowed from its [`Image`].
489type ImageBufferRef<'image> = ImageBuffer<Rgba<u8>, &'image [u8]>;
490
491/// Borrows the pixel data of an 8-bit RGBA image, or returns `None` if it has
492/// no data.
493fn image_buffer(image: &Image) -> Option<ImageBufferRef<'_>> {
494    let data = image.data.as_ref()?;
495    ImageBuffer::from_raw(image.width(), image.height(), data.as_slice())
496}

498/// The images glyphs on one font texture are drawn from.
499struct PageBuffers<'image> {
500    /// The font texture.
501    texture: ImageBufferRef<'image>,
502    /// The texture's outline layers, when the text is drawn with an outline
503    /// color and the texture has them.
504    outline_layers: Option<OutlineLayerBuffers<'image>>,
505}
506
507/// The outline and glyph layers of a font texture. See
508/// [`ImageFontOutlineLayers`](crate::ImageFontOutlineLayers).
509struct OutlineLayerBuffers<'image> {
510    /// The outline layer.
511    outline: ImageBufferRef<'image>,
512    /// The glyph layer.
513    glyph: ImageBufferRef<'image>,
514}
515
516/// Draws each placed glyph into the output image buffer, blending it over the
517/// glyphs drawn before it.
518///
519/// Glyphs are tinted with `colors.color`. When the text has an outline color,
520/// every outline is drawn first in that color, from the outline layers, and
521/// then the inside of every glyph on top, from the glyph layers.
522///
523/// # Parameters
524/// - `image_font`: The font asset containing glyph metadata and character maps.
525/// - `render_context`: The context providing glyph texture regions.
526/// - `placements`: Where to draw each glyph, from [`layout_glyphs`].
527/// - `output_image`: The image buffer to which glyphs are rendered.
528/// - `pages`: The images to draw from, one per font texture.
529/// - `colors`: The colors to draw the text in.
530fn render_glyphs_to_image(
531    image_font: &ImageFont,
532    render_context: &RenderContext<'_>,
533    placements: &[GlyphPlacement],
534    output_image: &mut ImageBuffer<Rgba<u8>, Vec<u8>>,
535    pages: &[PageBuffers<'_>],
536    colors: TextColors,
537) {
538    if let Some(outline_color) = colors.outline_color {
539        for placement in placements {
540            let page_index = image_font.atlas_character_map[&placement.character].page_index;
541            if let Some(ref layers) = pages[page_index].outline_layers {
542                draw_glyph(
543                    output_image,
544                    &layers.outline,
545                    render_context,
546                    placement,
547                    outline_color,
548                );
549            }
550        }
551    }
552
553    for placement in placements {
554        let page_index = image_font.atlas_character_map[&placement.character].page_index;
555        let page = &pages[page_index];
556        let source = page
557            .outline_layers
558            .as_ref()
559            .map_or(&page.texture, |layers| &layers.glyph);
560        draw_glyph(
561            output_image,
562            source,
563            render_context,
564            placement,
565            colors.color,
566        );
567    }
568}

570/// Draws a placed glyph from `source`, tinted with `tint`, blending it over
571/// the output image.
572fn draw_glyph(
573    output_image: &mut ImageBuffer<Rgba<u8>, Vec<u8>>,
574    source: &ImageBufferRef<'_>,
575    render_context: &RenderContext<'_>,
576    placement: &GlyphPlacement,
577    tint: Color,
578) {
579    let rect = render_context.glyph_rect(placement.character);
580    let glyph = source.view(rect.min.x, rect.min.y, rect.width(), rect.height());
581    let (x, y) = (placement.position.x, placement.position.y);
582
583    if tint == Color::WHITE {
584        imageops::overlay(output_image, &*glyph, x, y);
585    } else {
586        let tint = tint.to_linear();
587        let tinted = image::RgbaImage::from_fn(glyph.width(), glyph.height(), |x, y| {
588            tint_pixel(glyph.get_pixel(x, y), tint)
589        });
590        imageops::overlay(output_image, &tinted, x, y);
591    }
592}

594/// Multiplies an sRGB pixel by a tint in linear space, the way a sprite's
595/// color tints its image.
596fn tint_pixel(pixel: Rgba<u8>, tint: LinearRgba) -> Rgba<u8> {
597    let [red, green, blue, alpha] = pixel.0;
598    let linear = LinearRgba::from(Srgba::rgba_u8(red, green, blue, alpha));
599    let tinted = LinearRgba::new(
600        linear.red * tint.red,
601        linear.green * tint.green,
602        linear.blue * tint.blue,
603        linear.alpha * tint.alpha,
604    );
605    Rgba(Srgba::from(tinted).to_u8_array())
606}

608/// Prepares the rendering context for image font text rendering.
609///
610/// This function retrieves the required font asset, associated textures, and
611/// constructs a `RenderContext` for the given `ImageFontText`. It returns all
612/// the necessary data to render the text as an image, including the font,
613/// textures, and context with rendering configuration.
614///
615/// # Parameters
616/// - `image_font_text`: The text and font information to render.
617/// - `image_fonts`: The collection of available font assets.
618/// - `images`: The collection of image assets used to retrieve font textures.
619/// - `layouts`: The texture atlas layouts defining character positioning.
620///
621/// # Returns
622/// A tuple containing:
623/// - A reference to the `ImageFont` asset.
624/// - A vector of references to the font's texture images.
625/// - The constructed `RenderContext` for rendering.
626///
627/// # Errors
628/// Returns an error if the font asset or required texture assets are missing.
629fn prepare_context<'assets, 'image>(
630    image_font_text: &'assets ImageFontText,
631    image_fonts: &'assets Assets<ImageFont>,
632    images: &'image Assets<Image>,
633    layouts: &'assets Assets<TextureAtlasLayout>,
634) -> Result<
635    (
636        &'assets ImageFont,
637        Vec<&'image Image>,
638        RenderContext<'assets>,
639    ),
640    ImageFontRenderError,
641> {
642    let image_font = image_fonts
643        .get(&image_font_text.font)
644        .ok_or(ImageFontRenderError::MissingImageFontAsset)?;
645    let textures = image_font.textures(images);
646    let render_config = RenderConfig {
647        scaling_mode: ImageFontScalingMode::Truncated,
648        ..RenderConfig::default()
649    };
650    let render_context = RenderContext::new(image_font, image_font_text, render_config, layouts)
651        .ok_or(ImageFontRenderError::MissingTextureAsset)?;
652    Ok((image_font, textures, render_context))
653}

655/// Errors that can occur during the rendering of an `ImageFont`.
656#[derive(Debug, Error)]
657#[non_exhaustive]
658pub enum ImageFontRenderError {
659    /// The image could not be converted to a `DynamicImage`. This typically
660    /// indicates an issue with the underlying image data or its format.
661    #[error("failed to convert image to DynamicImage: {0}")]
662    ImageConversion(String),
663
664    /// The `ImageFont` asset required for rendering was not loaded. Ensure
665    /// that the font asset is correctly loaded into the Bevy app. This should
666    /// happend automatically when using the `AssetLoader` to load the font
667    /// asset.

690/// A helper trait that represents types that can hold an image handle.
691///
692/// This is used to abstract over different components (e.g., [`Sprite`] and
693/// [`ImageNode`]) that need to store a handle to an image rendered from text.
694trait ImageHandleHolder {
695    /// Sets the handle for the image that this holder represents.
696    ///
697    /// This method is called after rendering text into an image
698    /// to update the holder with the new image handle.
699    ///
700    /// # Parameters
701    /// - `image`: The handle to the newly rendered image.
702    fn set_image_handle(&mut self, image: Handle<Image>);
703}
704
705impl ImageHandleHolder for Sprite {
706    fn set_image_handle(&mut self, image: Handle<Image>) {
707        self.image = image;
708    }
709}
710
711#[cfg(feature = "ui")]
712impl ImageHandleHolder for ImageNode {
713    fn set_image_handle(&mut self, image: Handle<Image>) {
714        self.image = image;
715    }
716}

742        app: &App,
743        font: &Handle<ImageFont>,
744        text: &str,
745        font_height: Option<f32>,
746        colors: TextColors,
747    ) -> Result<RgbaImage, ImageFontRenderError> {
748        let world = app.world();

750            text: String::from(text),

756            colors,

758            world.resource::<Assets<Image>>(),

760        )?;
761        Ok(RgbaImage::from_raw(image.width(), image.height(), image.data.unwrap()).unwrap())
762    }

764    fn glyph(app: &App, font: &Handle<ImageFont>, character: char) -> RgbaImage {
765        let world = app.world();
766        let image_font = world.resource::<Assets<ImageFont>>().get(font).unwrap();
767        let image_font_character = &image_font.atlas_character_map[&character];
768        let rect = world
769            .resource::<Assets<TextureAtlasLayout>>()
770            .get(&image_font.atlas_layouts[image_font_character.page_index])
771            .unwrap()
772            .textures[image_font_character.character_index];
773        let texture = world
774            .resource::<Assets<Image>>()
775            .get(&image_font.textures[image_font_character.page_index])
776            .unwrap();
777        let texture = RgbaImage::from_raw(
778            texture.width(),
779            texture.height(),
780            texture.data.clone().unwrap(),
781        )
782        .unwrap();
783        imageops::crop_imm(
784            &texture,
785            rect.min.x,
786            rect.min.y,
787            rect.width(),
788            rect.height(),
789        )
790        .to_image()
791    }
