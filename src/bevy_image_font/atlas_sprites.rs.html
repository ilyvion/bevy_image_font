1//! This module provides functionality for rendering text as individual sprites
2//! using the Bevy engine, utilizing custom image fonts.
3//!
4//! It breaks down text into individual characters and represents them as
5//! sprites in the game world. This approach allows precise positioning and
6//! styling of text at the character level, suitable for scenarios where text
7//! needs to be rendered dynamically or interactively.
8//!
9//! Key Features:
10//! - `ImageFontSpriteText` component: Allows customization of text rendering,
11//!   such as color and anchor point.
12//! - Systems for rendering text to sprite entities and updating their
13//!   configuration when text changes.
14//! - Optional gizmo rendering for debugging purposes, available with the
15//!   "gizmos" feature flag.
16//!
17//! This module is intended for advanced text rendering use cases, offering
18//! fine-grained control over how text is displayed in the game world.
19

50/// Internal plugin for conveniently organizing the code related to this
51/// module's feature.
52#[derive(Default)]
53pub(crate) struct AtlasSpritesPlugin;
54
55impl Plugin for AtlasSpritesPlugin {
56    fn build(&self, app: &mut App) {
57        app.init_resource::<ExcessSprites>()
58            .add_systems(
59                PostUpdate,
60                set_up_sprites
61                    .after(sync_texts_with_font_changes)
62                    .before(TransformSystems::Propagate)
63                    .in_set(ImageFontSystems),
64            )
65            .add_systems(Last, despawn_excess_sprites.in_set(ImageFontSystems));
66
67        #[cfg(feature = "gizmos")]
68        {
69            gizmos::build(app);
70        }
71    }
72}

74/// Text rendered using an [`ImageFont`] as individual sprites.
75///
76/// This struct provides fields for customizing text rendering, such as
77/// alignment, color, and scaling behavior.
78///
79/// - `anchor`: Specifies the alignment point of the text relative to its
80///   position.
81/// - `color`: Uniform tint applied to all glyphs.
82/// - `scaling_mode`: Controls how scaling is applied to glyph dimensions.
83#[derive(Debug, Clone, Reflect, Default, Component, Setters)]
84#[setters(into)]
85#[require(ImageFontText, Visibility)]
86#[non_exhaustive]
87pub struct ImageFontSpriteText {
88    /// The alignment point of the text relative to its position. For example,
89    /// `Anchor::TopLeft` aligns the text's top-left corner to its position.
90    pub anchor: Anchor,
91
92    /// The color applied to the rendered text. This color affects all glyphs
93    /// equally, allowing you to tint the text uniformly.
94    pub color: Color,
95
96    /// The color of the outline, for fonts whose textures hold an outline
97    /// around their glyphs (see [`ImageFont::outline_layers`]).
98    ///
99    /// When `None`, the outline keeps the color it has in the font texture,
100    /// tinted by `color` like the rest of the glyph. When set, each glyph is
101    /// drawn as two sprites: the outline in this color, and the inside of the
102    /// glyph in `color` on top of it. Every outline sprite is placed slightly
103    /// behind the glyph sprites, so outlines never cover neighboring glyphs.
104    /// Fonts without an outline ignore this setting.
105    ///
106    /// The default value is `None`.
107    pub outline_color: Option<Color>,
108
109    /// Determines how scaling is applied to the glyph dimensions when adjusting
110    /// them to match the desired font height.
111    ///
112    /// This field allows control over how fractional scaling values are
113    /// handled, using the [`ImageFontScalingMode`] enum. It provides options to
114    /// truncate, round, or retain precise fractional values, depending on
115    /// your rendering requirements.
116    ///
117    /// The default value is `ImageFontScalingMode::Rounded`.
118    pub scaling_mode: ImageFontScalingMode,
119
120    /// Determines a constant kerning between characters. The spacing is given
121    /// at the font's native height and is scaled proportionally based on the
122    /// current font height.
123    pub letter_spacing: LetterSpacing,
124}

126/// Stores a mapping between characters and their corresponding sprite entities.
127/// This is used to manage text rendering at the entity level.
128#[derive(Debug, Clone, Component)]
129struct ImageFontTextData {
130    /// The entity that owns this `ImageFontTextData` component.
131    self_entity: Entity,
132
133    /// Basically a map between character index and character sprite
134    sprites: Vec<Entity>,
135
136    /// The outline sprites of the characters drawn with an outline color, in
137    /// text order.
138    outline_sprites: Vec<Entity>,
139
140    /// Tracks whether a missing font asset has already been reported for this
141    /// entity.
142    ///
143    /// This flag prevents repeated error messages when an `ImageFontText`
144    /// component references a font asset that has not been loaded. Once an
145    /// error is logged for a missing font, subsequent frames will not log
146    /// additional errors for the same entity until its font is found again.
147    ///
148    /// # Default Behavior
149    /// - Initially set to `false`, meaning no missing font error has been
150    ///   reported.
151    /// - Set to `true` after the first error message is logged.
152    /// - Set back to `false` whenever the font is found.
153    has_reported_missing_font: bool,
154
155    /// Stores debugging information for visualizing `ImageFontSpriteText` in
156    /// the scene.
157    ///
158    /// This field is only available when the `gizmos` feature is enabled. It
159    /// tracks glyph dimensions and other relevant data used for rendering
160    /// debug gizmos, such as bounding boxes and anchor points.
161    #[cfg(feature = "gizmos")]
162    gizmo_data: ImageFontTextGizmoData,
163}
164impl ImageFontTextData {
165    /// Creates a new `ImageFontTextData` instance for a given entity.
166    ///
167    /// This function initializes an empty `ImageFontTextData` struct and
168    /// associates it with the provided entity. It ensures that newly
169    /// created text data always has a reference to its owning entity.
170    ///
171    /// # Parameters
172    /// - `self_entity`: The entity that this `ImageFontTextData` belongs to.
173    ///
174    /// # Returns
175    /// A new instance of `ImageFontTextData` with an empty sprite list and the
176    /// specified `self_entity`.
177    fn new(entity: Entity) -> Self {
178        Self {
179            self_entity: entity,
180            sprites: Vec::default(),
181            outline_sprites: Vec::default(),
182            has_reported_missing_font: false,
183            #[cfg(feature = "gizmos")]
184            gizmo_data: ImageFontTextGizmoData::default(),
185        }
186    }
187}

189/// System that renders each [`ImageFontText`] as child [`Sprite`] entities,
190/// where each sprite represents a character in the text. Each sprite is
191/// positioned based on its order in the text, accounting for letter spacing,
192/// scaling mode, and anchor alignment. This system only runs when the
193/// `ImageFontText` or [`ImageFontSpriteText`] changes.
194#[expect(
195    clippy::missing_panics_doc,
196    reason = "expect() is only used on a newly created Some() value"
197)]
198#[expect(
199    private_interfaces,
200    reason = "Systems are only `pub` for the sake of allowing dependent crates to use them for ordering"
201)]
202pub fn set_up_sprites(
203    mut commands: Commands,
204    mut query: Query<
205        (
206            Entity,
207            &ImageFontText,
208            &ImageFontSpriteText,
209            Option<&mut ImageFontTextData>,
210        ),
211        Or<(Changed<ImageFontText>, Changed<ImageFontSpriteText>)>,
212    >,
213    mut child_query: Query<ChildSprite>,
214    mut excess_sprites: ResMut<ExcessSprites>,
215    image_fonts: Res<Assets<ImageFont>>,
216    texture_atlas_layouts: Res<Assets<TextureAtlasLayout>>,
217) {
218    for (entity, image_font_text, image_font_sprite_text, mut image_font_text_data) in &mut query {
219        let mut maybe_new_image_font_text_data = None;
220        let image_font_text_data = if let Some(image_font_text_data) = image_font_text_data.as_mut()
221        {
222            &mut *image_font_text_data
223        } else {
224            maybe_new_image_font_text_data = Some(ImageFontTextData::new(entity));
225            #[expect(clippy::expect_used, reason = "newly created Some() value")]
226            maybe_new_image_font_text_data
227                .as_mut()
228                .expect("newly created Some() value")
229        };
230
231        let render_config = RenderConfig {
232            letter_spacing: image_font_sprite_text.letter_spacing.to_f32(),
233            offset_characters: true,
234            apply_scaling: true,
235            scaling_mode: image_font_sprite_text.scaling_mode,
236            color: image_font_sprite_text.color,
237            outline_color: image_font_sprite_text.outline_color,
238            text_anchor: image_font_sprite_text.anchor,
239        };
240
241        let font_handle = &image_font_text.font;
242        let Some(image_font) = image_fonts.get(font_handle) else {
243            if !image_font_text_data.has_reported_missing_font {
244                let font_handle_detail: &dyn Debug = if let Some(font_path) = font_handle.path() {
245                    font_path
246                } else {
247                    &font_handle.id()
248                };
249                bevy_log::error!(
250                    "ImageFont asset {font_handle_detail:?} is not loaded; can't render text for entity: {}",
251                    image_font_text_data.self_entity
252                );
253                image_font_text_data.has_reported_missing_font = true;
254            }
255            maybe_insert_new_image_font_text_data(
256                &mut commands,
257                entity,
258                maybe_new_image_font_text_data,
259            );
260            continue;
261        };
262        image_font_text_data.has_reported_missing_font = false;
263
264        let Some(render_context) = RenderContext::new(
265            image_font,
266            image_font_text,
267            render_config,
268            &texture_atlas_layouts,
269        ) else {
270            maybe_insert_new_image_font_text_data(
271                &mut commands,
272                entity,
273                maybe_new_image_font_text_data,
274            );
275            continue;
276        };
277
278        let (glyph_sprites, outline_sprites) = text_sprites(&render_context, image_font_text_data);
279
280        let mut sprite_sync = SpriteSync {
281            commands: &mut commands,
282            parent: entity,
283            child_query: &mut child_query,
284            excess_sprites: &mut excess_sprites.0,
285        };
286        sprite_sync.sync(&mut image_font_text_data.sprites, glyph_sprites);
287        sprite_sync.sync(&mut image_font_text_data.outline_sprites, outline_sprites);
288
289        maybe_insert_new_image_font_text_data(
290            &mut commands,
291            entity,
292            maybe_new_image_font_text_data,
293        );
294    }
295}

297/// Inserts a newly created `ImageFontTextData` component into an entity if one
298/// was generated.
299///
300/// This function takes an `Option<ImageFontTextData>`, which may contain a
301/// newly created instance. If it is `Some`, the function inserts the
302/// `ImageFontTextData` into the provided entity. If it is `None`, no action is
303/// taken.
304///
305/// # Parameters
306/// - `commands`: A command buffer used to insert the component if
307///   `maybe_new_image_font_text_data` is `Some`.
308/// - `entity`: The entity that will receive the `ImageFontTextData`.
309/// - `maybe_new_image_font_text_data`: An `Option<ImageFontTextData>` that
310///   contains a newly created instance if one was generated earlier.
311///
312/// # Notes
313/// - This function does not check whether the entity already has an
314///   `ImageFontTextData` component; it simply inserts the given value if it
315///   exists.
316/// - If a new component is inserted, a debug message is logged.
317fn maybe_insert_new_image_font_text_data(
318    commands: &mut Commands,
319    entity: Entity,
320    maybe_new_image_font_text_data: Option<ImageFontTextData>,
321) {
322    if let Some(new_image_font_text_data) = maybe_new_image_font_text_data {
323        bevy_log::debug!("Inserted new ImageFontTextData for entity {:?}", entity);
324        commands.entity(entity).insert(new_image_font_text_data);
325    }
326}

328/// How far behind the glyph sprites the outline sprites are placed, so that
329/// every outline is drawn before every glyph.
330const OUTLINE_DEPTH: f32 = 0.001;
331
332/// Computes the sprites that draw the filtered text: one glyph sprite per
333/// character, and one outline sprite per character drawn with an outline
334/// color, placed [`OUTLINE_DEPTH`] behind its glyph sprite.
335///
336/// # Returns
337/// The glyph sprites and the outline sprites, each with its transform, in
338/// text order.
339fn text_sprites(
340    render_context: &RenderContext,
341    #[cfg_attr(
342        not(feature = "gizmos"),
343        expect(unused_variables, reason = "gizmos only")
344    )]
345    image_font_text_data: &mut ImageFontTextData,
346) -> (Vec<(Sprite, Transform)>, Vec<(Sprite, Transform)>) {
347    let mut glyph_sprites = Vec::new();
348    let mut outline_sprites = Vec::new();
349    #[cfg(feature = "gizmos")]
350    image_font_text_data.gizmo_data.clear();
351
352    for (line_index, line) in render_context.lines().iter().enumerate() {
353        let mut x_pos = 0.;
354        let mut characters = line.filtered_chars().peekable();
355        while let Some(character) = characters.next() {
356            let transform = render_context.sprite_transform(
357                line_index,
358                &mut x_pos,
359                character,
360                characters.peek().copied(),
361            );
362
363            if let Some(outline_sprite) = render_context.outline_sprite(character) {
364                let mut outline_transform = transform;
365                outline_transform.translation.z -= OUTLINE_DEPTH;
366                outline_sprites.push((outline_sprite, outline_transform));
367            }
368            glyph_sprites.push((render_context.glyph_sprite(character), transform));
369
370            #[cfg(feature = "gizmos")]
371            gizmos::record_character_dimensions(
372                render_context,
373                character,
374                &mut image_font_text_data.gizmo_data,
375            );
376        }
377    }
378
379    (glyph_sprites, outline_sprites)
380}

382/// The components of a text entity's child sprite that are kept in sync with
383/// its text.
384type ChildSprite = (&'static mut Sprite, &'static mut Transform);
385
386/// Sprites no longer used by their text, despawned by
387/// [`despawn_excess_sprites`].
388///
389/// They are despawned in [`Last`] because other `PostUpdate` systems, such as
390/// Bevy's sprite rendering, may queue commands for them that panic if they are
391/// applied after the sprite is despawned.
392#[derive(Debug, Default, Resource)]
393struct ExcessSprites(Vec<Entity>);
394
395/// Despawns the sprites in [`ExcessSprites`].
396fn despawn_excess_sprites(mut commands: Commands, mut excess_sprites: ResMut<ExcessSprites>) {
397    for entity in excess_sprites.0.drain(..) {
398        // The text entity may have been despawned along with its sprites.
399        commands.entity(entity).try_despawn();
400    }
401}

403/// Keeps a text entity's child sprites in sync with the sprites its text
404/// should be drawn with.
405struct SpriteSync<'borrow, 'world, 'state, 'query_world, 'query_state> {
406    /// Command buffer for spawning sprites.
407    commands: &'borrow mut Commands<'world, 'state>,
408    /// The text entity the sprites are children of.
409    parent: Entity,
410    /// Query for updating existing child sprites.
411    child_query: &'borrow mut Query<'query_world, 'query_state, ChildSprite>,
412    /// Where sprites that are no longer needed are collected for despawning.
413    excess_sprites: &'borrow mut Vec<Entity>,
414}
415
416impl SpriteSync<'_, '_, '_, '_, '_> {
417    /// Makes `entities` draw `sprites`: existing entities are updated in
418    /// order, missing ones are spawned as children of the text entity, and
419    /// excess ones are queued in [`ExcessSprites`] for despawning.
420    fn sync(&mut self, entities: &mut Vec<Entity>, sprites: Vec<(Sprite, Transform)>) {
421        self.excess_sprites
422            .extend(entities.drain(sprites.len().min(entities.len())..));
423
424        let mut sprites = sprites.into_iter();
425        for (&entity, (new_sprite, new_transform)) in entities.iter().zip(sprites.by_ref()) {
426            match self.child_query.get_mut(entity) {
427                Ok((mut sprite, mut transform)) => {
428                    sprite.image = new_sprite.image;
429                    sprite.texture_atlas = new_sprite.texture_atlas;
430                    sprite.color = new_sprite.color;
431                    *transform = new_transform;
432                }
433                Err(error) => {
434                    bevy_log::error!(
435                        "An ImageFontSpriteText unexpectedly failed: {error}. This will likely \
436                        cause rendering bugs."
437                    );
438                }
439            }
440        }
441
442        self.commands.entity(self.parent).with_children(|parent| {
443            for sprite in sprites {
444                entities.push(parent.spawn(sprite).id());
445            }
446        });
447    }
448}
