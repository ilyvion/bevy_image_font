1//! Debug visualization for `ImageFontSpriteText` components.
2//!
3//! This module provides systems and configurations for rendering **debug
4//! gizmos** that help visualize the alignment, positioning, and bounding boxes
5//! of `ImageFontSpriteText` elements in the scene.
6//!
7//! # Features
8//! - **Anchor Point Visualization**: Draws small crosshairs to indicate text
9//!   and character anchor positions.
10//! - **Bounding Box Visualization**: Renders rectangles around individual
11//!   glyphs to assist with alignment debugging.
12//! - **Configurable Rendering**: Gizmos can be toggled globally via
13//!   [`AtlasSpritesGizmoConfigGroup`] or on a per-entity basis using
14//!   [`ShowAtlasSpritesGizmos`].
15

37/// Initializes the debug gizmo system for `ImageFontSpriteText` components.
38///
39/// This function sets up the necessary configuration and systems to enable
40/// debug visualization of `ImageFontSpriteText` elements in Bevy's scene.
41/// It registers the `AtlasSpritesGizmoConfigGroup` and adds the rendering
42/// systems for drawing sprite-based text gizmos.
43///
44/// # Parameters
45/// - `app`: A mutable reference to the [`App`] to which the gizmo systems are
46///   added.
47///
48/// # Behavior
49/// - Registers `AtlasSpritesGizmoConfigGroup`, which controls whether various
50///   text gizmos (such as anchor points and bounding boxes) are rendered.
51/// - Adds the `render_all_sprite_gizmos` and `render_sprite_gizmos` systems to
52///   the update schedule, enabling debug visualization.
53pub fn build(app: &mut App) {
54    app.init_gizmo_group::<AtlasSpritesGizmoConfigGroup>()
55        .add_systems(
56            Update,
57            (render_all_sprite_gizmos, render_sprite_gizmos).in_set(ImageFontSet),
58        );
59}

61/// The [`GizmoConfigGroup`] used for debug visualizations of
62/// [`ImageFontSpriteText`](crate::atlas_sprites::ImageFontSpriteText)
63/// components on entities
64#[derive(Debug, GizmoConfigGroup, Reflect, Resource)]
65pub struct AtlasSpritesGizmoConfigGroup {
66    /// Draws text anchor points in the scene when set to `true`.
67    ///
68    /// To draw a specific entity's text anchor point, you can add the
69    /// [`ShowAtlasSpritesGizmos`] component.
70    ///
71    /// Defaults to `false`.
72    pub render_text_anchor_point: bool,
73
74    /// The color to draws the text anchor points in the scene with.
75    ///
76    /// Defaults to `css::RED`.
77    pub text_anchor_point_color: Color,
78
79    /// Draws character anchor points in the scene when set to `true`.
80    ///
81    /// To draw a specific entity's character anchor points, you can add the
82    /// [`ShowAtlasSpritesGizmos`] component.
83    ///
84    /// Defaults to `false`.
85    pub render_character_anchor_point: bool,
86
87    /// The color to draws the character anchor points in the scene with.
88    ///
89    /// Defaults to `css::GREEN`.
90    pub character_anchor_point_color: Color,
91
92    /// Draws character bounding boxes in the scene when set to `true`.
93    ///
94    /// To draw a specific entity's character bounding boxes, you can add the
95    /// [`ShowAtlasSpritesGizmos`] component.
96    ///
97    /// Defaults to `false`.
98    pub render_character_box: bool,
99
100    /// The color to draws the character bounding boxes in the scene with.
101    ///
102    /// Defaults to `css::PURPLE`.
103    pub character_box_color: Color,
104}
105
106impl Default for AtlasSpritesGizmoConfigGroup {
107    fn default() -> Self {
108        Self {
109            render_text_anchor_point: Default::default(),
110            text_anchor_point_color: css::RED.into(),
111            render_character_anchor_point: Default::default(),
112            character_anchor_point_color: css::GREEN.into(),
113            render_character_box: Default::default(),
114            character_box_color: css::PURPLE.into(),
115        }
116    }
117}

174/// Debugging data for visualizing an `ImageFontSpriteText` in a scene, enabled
175/// by the `gizmos` feature.
176#[derive(Debug, Clone, Default, Component)]
177pub struct ImageFontTextGizmoData {
178    /// The bounding box size of each glyph sprite, in the same order as the
179    /// glyph sprites of the text.
180    glyph_sizes: Vec<Vec2>,
181}
182
183impl ImageFontTextGizmoData {
184    /// Forgets the recorded glyph sizes, before the sizes of a new layout are
185    /// recorded.
186    pub(crate) fn clear(&mut self) {
187        self.glyph_sizes.clear();
188    }
189}
190
191/// Records the width and height of the next glyph sprite for gizmo
192/// visualization.
193///
194/// This function queries the `RenderContext` for the dimensions of the given
195/// character and appends them to the provided `ImageFontTextGizmoData`.
196pub(crate) fn record_character_dimensions(
197    render_context: &RenderContext,
198    character: char,
199    gizmo_data: &mut ImageFontTextGizmoData,
200) {
201    let (width, height) = render_context.character_dimensions(character);
202    gizmo_data.glyph_sizes.push(Vec2::new(width, height));
203}
204
205/// The position and bounding box size of each glyph sprite of a text.
206pub(super) fn glyph_boxes<'data>(
207    data: &'data ImageFontTextData,
208    child_query: &'data Query<&GlobalTransform, Without<ImageFontText>>,
209) -> impl Iterator<Item = (Vec2, Vec2)> + 'data {
210    data.sprites
211        .iter()
212        .zip(&data.gizmo_data.glyph_sizes)
213        .filter_map(|(&sprite, &size)| {
214            let transform = child_query.get(sprite).ok()?;
215            Some((transform.translation().truncate(), size))
216        })
217}

219/// Renders gizmos for debugging `ImageFontText` and its associated glyphs in
220/// the scene.
221///
222/// This function draws 2D rectangles and crosshairs to visualize the bounding
223/// boxes and positions of rendered glyphs, aiding in debugging and alignment.
224///
225/// ### Gizmo Details
226/// - Each child glyph is visualized as a purple rectangle using its dimensions
227///   and position.
228/// - The `ImageFontText` position is marked with a red cross for easier
229///   identification.
230fn render_all_sprite_gizmos(
231    mut gizmos: Gizmos<AtlasSpritesGizmoConfigGroup>,
232    query: Query<
233        (&GlobalTransform, &ImageFontTextData),
234        (With<ImageFontText>, Without<ShowAtlasSpritesGizmos>),
235    >,
236    child_query: Query<&GlobalTransform, Without<ImageFontText>>,
237) {
238    for (global_transform, data) in &query {
239        render_gizmos(&mut gizmos, &child_query, None, global_transform, data);
240    }
241}

246/// This function draws 2D rectangles and crosshairs to visualize the bounding

249/// ### Gizmo Details
250/// - Each child glyph is visualized as a purple rectangle using its dimensions

258            &ShowAtlasSpritesGizmos,

271            global_transform,
272            data,
273        );
274    }

277/// Retrieves a gizmo configuration value, prioritizing per-entity settings.
278///
279/// This macro allows a [`ShowAtlasSpritesGizmos`] component (if present) to
280/// override the default configuration from [`AtlasSpritesGizmoConfigGroup`].
281///
282/// # Parameters
283/// - `$gizmos`: A reference to the [`Gizmos<AtlasSpritesGizmoConfigGroup>`]
284///   instance.
285/// - `$sprite_gizmos`: An `Option<&ShowAtlasSpritesGizmos>` for the entity.
286/// - `$field`: The field name to retrieve.
287///
288/// # Behavior
289/// - If the entity has a [`ShowAtlasSpritesGizmos`] component **and** the field
290///   is set (`Some(value)`), the entity-specific value is used.
291/// - Otherwise, it falls back to the global configuration in
292///   [`AtlasSpritesGizmoConfigGroup`].
293macro_rules! gizmo_config_value {
294    ($gizmos:expr_2021, $sprite_gizmos:expr_2021, $field:ident) => {
295        $sprite_gizmos
296            .and_then(|sprite_gizmos| sprite_gizmos.$field)
297            .unwrap_or($gizmos.config_ext.$field)
298    };
299}

301/// Renders debug gizmos for a single `ImageFontText` entity and its glyph
302/// sprites.
303///
304/// This function draws rectangles and crosshairs to visualize the bounding
305/// boxes and anchor points of each glyph sprite and the text entity itself,
306/// according to the current gizmo configuration. Per-entity overrides are
307/// supported via `ShowAtlasSpritesGizmos`.
308fn render_gizmos(
309    gizmos: &mut Gizmos<AtlasSpritesGizmoConfigGroup>,
310    child_query: &Query<&GlobalTransform, Without<ImageFontText>>,
311    sprite_gizmos: Option<&ShowAtlasSpritesGizmos>,
312    global_transform: &GlobalTransform,
313    data: &ImageFontTextData,
314) {
315    for (position, size) in glyph_boxes(data, child_query) {
316        if gizmo_config_value!(gizmos, sprite_gizmos, render_character_box) {
317            let color = gizmo_config_value!(gizmos, sprite_gizmos, character_box_color);
318            gizmos.rect_2d(
319                bevy_math::Isometry2d::from_translation(position),
320                size,
321                color,
322            );
323        }
324
325        if gizmo_config_value!(gizmos, sprite_gizmos, render_character_anchor_point) {
326            let color = gizmo_config_value!(gizmos, sprite_gizmos, character_anchor_point_color);
327            gizmos.cross_2d(bevy_math::Isometry2d::from_translation(position), 5., color);
328        }
329    }
330
331    if gizmo_config_value!(gizmos, sprite_gizmos, render_text_anchor_point) {
332        let color = gizmo_config_value!(gizmos, sprite_gizmos, text_anchor_point_color);
333        gizmos.cross_2d(
334            bevy_math::Isometry2d::from_translation(global_transform.translation().truncate()),
335            10.,
336            color,
337        );
338    }
339}
