31/// An error type representing issues that may arise during the loading of BMF
32/// fonts.
33///
34/// This includes errors encountered when processing the font file, handling
35/// associated assets, or enforcing constraints specific to the
36/// [`BmFontLoader`]. Some errors originate from lower-level font loading
37/// mechanisms and are wrapped transparently.
38#[derive(Debug, Error)]
39#[non_exhaustive]
40pub enum BmFontLoadError {
41    /// Errors that can occur when loading an image font.
42    #[error(transparent)]
43    ImageFontLoadError(#[from] ImageFontLoadError),
44
45    /// Errors that occur while parsing the BMF file format.
46    #[error(transparent)]
47    BmFontError(#[from] bmfont_rs::Error),
48
49    /// The loaded font is not a Unicode font, and the `bmf_charsets` feature,
50    /// which [`BmFontLoader`] needs to load such fonts, is not enabled.
51    #[error("BmFontLoader only supports unicode fonts without the `bmf_charsets` feature")]
52    CharsetUnsupported,
53
54    /// The loaded font is not a Unicode font, and its `info.charset` is not a
55    /// charset `BMFont` defines.
56    #[cfg(feature = "bmf_charsets")]
57    #[error("unknown BMFont charset {0:?}")]
58    UnknownCharset(String),
59
60    /// The loaded font is not a Unicode font, and the code page its charset
61    /// stands for can't be decoded. The `JOHAB` charset (code page 1361) is
62    /// not supported, and neither are code pages in
63    /// [`ImageFontLoaderSettings::bmf_ansi_code_page`] or
64    /// [`ImageFontLoaderSettings::bmf_oem_code_page`] that neither
65    /// `encoding_rs` nor `oem_cp` can decode.
66    #[cfg(feature = "bmf_charsets")]
67    #[error("BMFont code page {0} is not supported")]
68    CodePageUnsupported(u16),
69}

71impl AssetLoader for BmFontLoader {
72    type Asset = ImageFont;
73
74    type Settings = ImageFontLoaderSettings;
75
76    type Error = BmFontLoadError;
77
78    async fn load(
79        &self,
80        reader: &mut dyn Reader,
81        settings: &Self::Settings,
82        load_context: &mut LoadContext<'_>,
83    ) -> Result<Self::Asset, Self::Error> {
84        let data = read_bmf_data(reader).await?;
85        let bm_font = parse_bmf_data(&data, load_context)?;
86        let char_ids = CharIds::new(&bm_font.info, settings)?;
87
88        let mut atlas_character_map = process_bmf_characters(&bm_font, char_ids);
89        let textures =
90            load_images_and_textures(&bm_font, &mut atlas_character_map, settings, load_context)
91                .await?;
92
93        Ok(construct_image_font(
94            &bm_font,
95            char_ids,
96            textures,
97            atlas_character_map,
98            settings,
99        ))
100    }
101
102    fn extensions(&self) -> &[&str] {
103        BmFontExtension::VARIANTS
104    }
105}

107/// Represents the supported file extensions for BMF font files.
108///
109/// This enum is used to ensure that only recognized file formats are parsed
110/// correctly, preventing unsupported formats from being loaded.
111#[derive(AsRefStr, Debug, Copy, Clone, EnumIter, PartialEq, Eq, VariantNames)]
112#[repr(usize)]
113enum BmFontExtension {
114    /// The text-based `.txt.fnt` format.
115    ///
116    /// This format stores font metadata in a human-readable text format.
117    #[strum(serialize = "txt.fnt")]
118    Text,
119
120    /// The XML-based `.xml.fnt` format.
121    ///
122    /// This format stores font metadata using XML syntax.
123    #[strum(serialize = "xml.fnt")]
124    Xml,
125
126    /// The binary `.bin.fnt` format.
127    ///
128    /// This format stores font metadata in a compact binary representation.
129    #[strum(serialize = "bin.fnt")]
130    Binary,
131}
132
133impl BmFontExtension {
134    /// Attempts to convert an extension tuple (`(&str, &str)`) into a
135    /// `BmFontExtension`.
136    ///
137    /// # Parameters
138    /// - `extension`: A tuple representing the two-part file extension, e.g.,
139    ///   `("txt", "fnt")`.
140    ///
141    /// # Returns
142    /// - `Some(BmFontExtension)` if the extension is recognized.
143    /// - `None` if the extension does not match any known format.
144    #[inline]
145    fn from_tuple(extension: (&str, &str)) -> Option<Self> {
146        Self::iter().find(|&self_extension| extension == self_extension.as_tuple())
147    }
148
149    /// Extracts the `BmFontExtension` from a `AssetPath`, if possible.
150    ///
151    /// This function looks at the file name, extracts its two-dot extension,
152    /// and attempts to match it against known BMF font extensions.
153    ///
154    /// # Parameters
155    /// - `path`: A reference to an `AssetPath` representing the file path.
156    ///
157    /// # Returns
158    /// - `Some(BmFontExtension)` if the file has a recognized extension.
159    /// - `None` if the file's extension is not recognized or the path has no
160    ///   valid file name.
161    fn from_asset_path(path: &AssetPath) -> Option<Self> {
162        let file_name = path.get_full_extension()?;
163        let (base, ext2) = file_name.rsplit_once('.')?;
164        let ext1 = base.rsplit_once('.').map_or(base, |(_, ext)| ext);
165        Self::from_tuple((ext1, ext2))
166    }
167
168    /// Converts a `BmFontExtension` into its corresponding two-dot extension
169    /// tuple.
170    ///
171    /// # Returns
172    /// A tuple containing the two parts of the extension, e.g., `("txt",
173    /// "fnt")`.
174    fn as_tuple(&self) -> (&str, &str) {
175        match self.as_ref().split_once('.') {
176            Some(tuple) => tuple,
177            None => unreachable!("Every BmFontExtension must contain a '.'"),
178        }
179    }
180}

182/// Reads the BMF font data from the reader.
183async fn read_bmf_data(reader: &mut dyn Reader) -> Result<Vec<u8>, ImageFontLoadError> {
184    let mut data = Vec::new();
185    reader.read_to_end(&mut data).await?;
186    Ok(data)
187}

189/// Parses the BMF font data.
190fn parse_bmf_data(
191    data: &[u8],
192    load_context: &mut LoadContext<'_>,
193) -> Result<bmfont_rs::Font, BmFontLoadError> {
194    let path = load_context.path();
195
196    let from_bytes = match BmFontExtension::from_asset_path(path) {
197        Some(BmFontExtension::Text) => bmfont_rs::text::from_bytes,
198        Some(BmFontExtension::Xml) => bmfont_rs::xml::from_bytes,
199        Some(BmFontExtension::Binary) => bmfont_rs::binary::from_bytes,
200        extension => unreachable!("{extension:?}"),
201    };
202
203    Ok(from_bytes(data)?)
204}

206/// A BMF character and where its glyph is in the font's textures.
207#[derive(Debug)]
208struct AtlasCharacter<'font> {
209    /// The character's definition in the font file.
210    font_char: &'font bmfont_rs::Char,
211    /// The glyph's rectangle on its page.
212    rect: URect,
213    /// The index of the glyph's texture and its index in that texture's atlas
214    /// layout, set by `load_images_and_textures`.
215    placement: Option<(usize, usize)>,
216}
217
218/// Processes BMF characters into an atlas character map.
219fn process_bmf_characters(
220    bm_font: &bmfont_rs::Font,
221    char_ids: CharIds,
222) -> HashMap<char, AtlasCharacter<'_>> {
223    let mut atlas_character_map = HashMap::new();
224    for char in &bm_font.chars {
225        let x = u32::from(char.x);
226        let y = u32::from(char.y);
227
228        let rect = URect {
229            min: UVec2 { x, y },
230            max: UVec2 {
231                x: x + u32::from(char.width),
232                y: y + u32::from(char.height),
233            },
234        };
235
236        if let Some(character) = char_ids.to_char(char.id) {
237            atlas_character_map.insert(
238                character,
239                AtlasCharacter {
240                    font_char: char,
241                    rect,
242                    placement: None,
243                },
244            );
245        } else {
246            warn!(
247                "Skipping invalid character id {}. Full char definition: {char:?} ",
248                char.id
249            );
250        }
251    }
252
253    atlas_character_map
254}

256/// The textures of a loaded font, with their atlas layouts and outline layers
257/// at the same indices.
258#[derive(Debug, Default)]
259struct FontTextures {
260    /// The font's textures.
261    images: Vec<Handle<Image>>,
262    /// The atlas layout of each texture.
263    atlas_layouts: Vec<Handle<TextureAtlasLayout>>,
264    /// The outline and glyph layers of each texture that holds an outline.
265    outline_layers: Vec<Option<ImageFontOutlineLayers>>,
266}

268/// Loads font images and creates texture atlases.
269///
270/// Grayscale pages are expanded to RGBA, and each page is decoded according to
271/// the font's channel settings. A packed
272/// page becomes one texture per channel its glyphs are in. Textures that hold
273/// an outline are also split into outline and glyph layers.
274async fn load_images_and_textures(
275    bm_font: &bmfont_rs::Font,
276    atlas_character_map: &mut HashMap<char, AtlasCharacter<'_>>,
277    settings: &ImageFontLoaderSettings,
278    load_context: &mut LoadContext<'_>,
279) -> Result<FontTextures, BmFontLoadError> {
280    let channel_settings = ChannelSettings::new(&bm_font.common);
281    let mut textures = FontTextures::default();
282    for (page_no, page) in bm_font.pages.iter().enumerate() {
283        let image_path = load_context
284            .path()
285            .resolve_embed_str(page)
286            .map_err(ImageFontLoadError::from)?;
287
288        let mut image: Image = load_context
289            .load_builder()
290            .with_settings(|settings: &mut ImageLoaderSettings| {
291                settings.expand_grayscale = false;
292            })
293            .load_value(image_path.clone())
294            .await
295            .map_err(ImageFontLoadError::from)?
296            .take();
297
298        image.sampler = settings.image_sampler.clone();
299        let channel_settings = match expand_grayscale_page(&mut image, bm_font.common.alpha_chnl) {
300            Some(alpha) => channel_settings.for_8_bit_page(alpha),
301            None => channel_settings,
302        };
303
304        let page_characters = atlas_character_map
305            .values_mut()
306            .filter(|character| usize::from(character.font_char.page) == page_no);
307        let mut views: Vec<(PageDecoding, Vec<&mut AtlasCharacter<'_>>)> = Vec::new();
308        if let ChannelSettings::NotPacked(decoding) = channel_settings {
309            views.push((decoding, page_characters.collect()));
310        } else {
311            for character in page_characters {
312                let decoding = channel_settings.decoding_for(character.font_char.chnl);
313                match views.iter_mut().find(|&&mut (view, _)| view == decoding) {
314                    Some(&mut (_, ref mut characters)) => characters.push(character),
315                    None => views.push((decoding, vec![character])),
316                }
317            }
318        }
319
320        let size = image.size();
321        let view_count = views.len();
322        let mut page_image = Some(image);
323        for (view_no, (decoding, characters)) in views.into_iter().enumerate() {
324            let mut view_image = if view_no + 1 == view_count {
325                page_image.take()
326            } else {
327                page_image.clone()
328            }
329            .unwrap_or_else(|| unreachable!("the page image is taken by the last view only"));
330            decoding.decode_image(&mut view_image);
331
332            let suffix = if channel_settings.is_packed() {
333                decoding.packed_label_suffix()
334            } else {
335                ""
336            };
337            let label = format!("texture_{page_no}{suffix}");
338            let layers = decoding
339                .has_outline()
340                .then(|| outline_layers(&view_image))
341                .flatten()
342                .map(|(outline, glyph)| ImageFontOutlineLayers {
343                    outline: load_context.add_labeled_asset(format!("{label}_outline"), outline),
344                    glyph: load_context.add_labeled_asset(format!("{label}_glyph"), glyph),
345                });
346            textures.outline_layers.push(layers);
347
348            let texture_index = textures.images.len();
349            textures
350                .images
351                .push(load_context.add_labeled_asset(label, view_image));
352
353            let mut atlas_layout = TextureAtlasLayout::new_empty(size);
354            for character in characters {
355                character.placement =
356                    Some((texture_index, atlas_layout.add_texture(character.rect)));
357            }
358            textures.atlas_layouts.push(
359                load_context.add_labeled_asset(format!("layout_{page_no}{suffix}"), atlas_layout),
360            );
361        }
362    }
363
364    Ok(textures)
365}

367/// Constructs the final `ImageFont` asset.
368fn construct_image_font(
369    bm_font: &bmfont_rs::Font,
370    char_ids: CharIds,
371    textures: FontTextures,
372    atlas_character_map: HashMap<char, AtlasCharacter<'_>>,
373    settings: &ImageFontLoaderSettings,
374) -> ImageFont {
375    ImageFont {
376        textures: textures.images,
377        atlas_character_map: atlas_character_map
378            .into_iter()
379            .map(|(char, character)| {
380                let font_char = character.font_char;
381                let kernings = bm_font.kernings.iter().filter_map(|kerning| {
382                    if kerning.first == font_char.id {
383                        if let Some(character) = char_ids.to_char(kerning.second) {
384                            Some((character, f32::from(kerning.amount)))
385                        } else {
386                            warn!(
387                                "Skipping invalid character id {} in kerning data.",
388                                kerning.second
389                            );
390                            None
391                        }
392                    } else {
393                        None
394                    }
395                });
396
397                (
398                    char,
399                    #[expect(
400                        clippy::unwrap_used,
401                        reason = "all Nones replaced in load_images_and_textures"
402                    )]
403                    ImageFontCharacter {
404                        character_index: character.placement.unwrap().1,
405                        page_index: character.placement.unwrap().0,
406                        offsets: Vec2::new(
407                            f32::from(font_char.xoffset),
408                            -f32::from(font_char.yoffset),
409                        ),
410                        x_advance: Some(f32::from(font_char.xadvance)),
411                        kernings: kernings.collect(),
412                    },
413                )
414            })
415            .collect(),
416        atlas_layouts: textures.atlas_layouts,
417        image_sampler: settings.image_sampler.clone(),
418        line_height: Some(u32::from(bm_font.common.line_height)),
419        outline_layers: textures.outline_layers,
420    }
