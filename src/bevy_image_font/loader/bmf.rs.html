31/// An error type representing issues that may arise during the loading of BMF
32/// fonts.
33///
34/// This includes errors encountered when processing the font file, handling
35/// associated assets, or enforcing constraints specific to the
36/// [`BmFontLoader`]. Some errors originate from lower-level font loading
37/// mechanisms and are wrapped transparently.
38#[derive(Debug, Error)]
39#[non_exhaustive]
40pub enum BmFontLoadError {
41    /// Errors that can occur when loading an image font.
42    #[error(transparent)]
43    ImageFontLoadError(#[from] ImageFontLoadError),
44
45    /// Errors that occur while parsing the BMF file format.
46    #[error(transparent)]
47    BmFontError(#[from] bmfont_rs::Error),
48
49    /// The loaded font is not a Unicode font, and the `bmf_charsets` feature,
50    /// which [`BmFontLoader`] needs to load such fonts, is not enabled.
51    #[error("BmFontLoader only supports unicode fonts without the `bmf_charsets` feature")]
52    CharsetUnsupported,
53
54    /// The font file's extension is not one of `txt.fnt`, `xml.fnt` or
55    /// `bin.fnt`, which [`BmFontLoader`] uses to tell the file's format.
56    #[error(
57        "BmFontLoader can't tell the format of '{0}'; its extension must be `txt.fnt`, `xml.fnt` \
58        or `bin.fnt`"
59    )]
60    UnsupportedExtension(AssetPath<'static>),
61
62    /// The loaded font is not a Unicode font, and its `info.charset` is not a
63    /// charset `BMFont` defines.
64    #[cfg(feature = "bmf_charsets")]
65    #[error("unknown BMFont charset {0:?}")]
66    UnknownCharset(String),
67
68    /// The loaded font is not a Unicode font, and the code page its charset
69    /// stands for can't be decoded. The `JOHAB` charset (code page 1361) is
70    /// not supported, and neither are code pages in
71    /// [`ImageFontLoaderSettings::bmf_ansi_code_page`] or
72    /// [`ImageFontLoaderSettings::bmf_oem_code_page`] that neither
73    /// `encoding_rs` nor `oem_cp` can decode.
74    #[cfg(feature = "bmf_charsets")]
75    #[error("BMFont code page {0} is not supported")]
76    CodePageUnsupported(u16),
77
78    /// A character's rectangle extends past the edges of its page's image.
79    #[error(
80        "The rectangle {rect:?} of character '{character}' does not fit inside the {}x{} image of \
81        page {page}", page_size.x, page_size.y
82    )]
83    CharacterRectOutOfBounds {
84        /// The character whose rectangle is out of bounds.
85        character: char,
86        /// The out-of-bounds rectangle.
87        rect: URect,
88        /// The index of the character's page.
89        page: usize,
90        /// The size of the page's image.
91        page_size: UVec2,
92    },
93}

95impl AssetLoader for BmFontLoader {
96    type Asset = ImageFont;
97
98    type Settings = ImageFontLoaderSettings;
99
100    type Error = BmFontLoadError;
101
102    async fn load(
103        &self,
104        reader: &mut dyn Reader,
105        settings: &Self::Settings,
106        load_context: &mut LoadContext<'_>,
107    ) -> Result<Self::Asset, Self::Error> {
108        let data = read_bmf_data(reader).await?;
109        let bm_font = parse_bmf_data(&data, load_context)?;
110        let char_ids = CharIds::new(&bm_font.info, settings)?;
111
112        let mut atlas_character_map = process_bmf_characters(&bm_font, char_ids);
113        let textures =
114            load_images_and_textures(&bm_font, &mut atlas_character_map, settings, load_context)
115                .await?;
116
117        Ok(construct_image_font(
118            &bm_font,
119            char_ids,
120            textures,
121            atlas_character_map,
122            settings,
123        ))
124    }
125
126    fn extensions(&self) -> &[&str] {
127        BmFontExtension::VARIANTS
128    }
129}

131/// Represents the supported file extensions for BMF font files.
132///
133/// This enum is used to ensure that only recognized file formats are parsed
134/// correctly, preventing unsupported formats from being loaded.
135#[derive(AsRefStr, Debug, Copy, Clone, EnumIter, PartialEq, Eq, VariantNames)]
136#[repr(usize)]
137enum BmFontExtension {
138    /// The text-based `.txt.fnt` format.
139    ///
140    /// This format stores font metadata in a human-readable text format.
141    #[strum(serialize = "txt.fnt")]
142    Text,
143
144    /// The XML-based `.xml.fnt` format.
145    ///
146    /// This format stores font metadata using XML syntax.
147    #[strum(serialize = "xml.fnt")]
148    Xml,
149
150    /// The binary `.bin.fnt` format.
151    ///
152    /// This format stores font metadata in a compact binary representation.
153    #[strum(serialize = "bin.fnt")]
154    Binary,
155}
156
157impl BmFontExtension {
158    /// Attempts to convert an extension tuple (`(&str, &str)`) into a
159    /// `BmFontExtension`.
160    ///
161    /// # Parameters
162    /// - `extension`: A tuple representing the two-part file extension, e.g.,
163    ///   `("txt", "fnt")`.
164    ///
165    /// # Returns
166    /// - `Some(BmFontExtension)` if the extension is recognized.
167    /// - `None` if the extension does not match any known format.
168    #[inline]
169    fn from_tuple(extension: (&str, &str)) -> Option<Self> {
170        Self::iter().find(|&self_extension| extension == self_extension.as_tuple())
171    }
172
173    /// Extracts the `BmFontExtension` from a `AssetPath`, if possible.
174    ///
175    /// This function looks at the file name, extracts its two-dot extension,
176    /// and attempts to match it against known BMF font extensions.
177    ///
178    /// # Parameters
179    /// - `path`: A reference to an `AssetPath` representing the file path.
180    ///
181    /// # Returns
182    /// - `Some(BmFontExtension)` if the file has a recognized extension.
183    /// - `None` if the file's extension is not recognized or the path has no
184    ///   valid file name.
185    fn from_asset_path(path: &AssetPath) -> Option<Self> {
186        let file_name = path.get_full_extension()?;
187        let (base, ext2) = file_name.rsplit_once('.')?;
188        let ext1 = base.rsplit_once('.').map_or(base, |(_, ext)| ext);
189        Self::from_tuple((ext1, ext2))
190    }
191
192    /// Converts a `BmFontExtension` into its corresponding two-dot extension
193    /// tuple.
194    ///
195    /// # Returns
196    /// A tuple containing the two parts of the extension, e.g., `("txt",
197    /// "fnt")`.
198    fn as_tuple(&self) -> (&str, &str) {
199        match self.as_ref().split_once('.') {
200            Some(tuple) => tuple,
201            None => unreachable!("Every BmFontExtension must contain a '.'"),
202        }
203    }
204}

206/// Reads the BMF font data from the reader.
207async fn read_bmf_data(reader: &mut dyn Reader) -> Result<Vec<u8>, ImageFontLoadError> {
208    let mut data = Vec::new();
209    reader.read_to_end(&mut data).await?;
210    Ok(data)
211}

213/// Parses the BMF font data.
214fn parse_bmf_data(
215    data: &[u8],
216    load_context: &mut LoadContext<'_>,
217) -> Result<bmfont_rs::Font, BmFontLoadError> {
218    let path = load_context.path();
219
220    let from_bytes = match BmFontExtension::from_asset_path(path) {
221        Some(BmFontExtension::Text) => bmfont_rs::text::from_bytes,
222        Some(BmFontExtension::Xml) => bmfont_rs::xml::from_bytes,
223        Some(BmFontExtension::Binary) => bmfont_rs::binary::from_bytes,
224        None => return Err(BmFontLoadError::UnsupportedExtension(path.clone_owned())),
225    };
226
227    let bm_font = from_bytes(data)?;
228    if let Some(char) = bm_font
229        .chars
230        .iter()
231        .find(|char| usize::from(char.page) >= bm_font.pages.len())
232    {
233        return Err(bmfont_rs::Error::InvalidCharPage {
234            char_id: char.id,
235            page_id: u32::from(char.page),
236        }
237        .into());
238    }
239
240    Ok(bm_font)
241}

243/// A BMF character and where its glyph is in the font's textures.
244#[derive(Debug)]
245struct AtlasCharacter<'font> {
246    /// The character's definition in the font file.
247    font_char: &'font bmfont_rs::Char,
248    /// The glyph's rectangle on its page.
249    rect: URect,
250    /// The index of the glyph's texture and its index in that texture's atlas
251    /// layout, set by `load_images_and_textures`.
252    placement: Option<(usize, usize)>,
253}
254
255/// Processes BMF characters into an atlas character map.
256fn process_bmf_characters(
257    bm_font: &bmfont_rs::Font,
258    char_ids: CharIds,
259) -> HashMap<char, AtlasCharacter<'_>> {
260    let mut atlas_character_map = HashMap::new();
261    for char in &bm_font.chars {
262        let x = u32::from(char.x);
263        let y = u32::from(char.y);
264
265        let rect = URect {
266            min: UVec2 { x, y },
267            max: UVec2 {
268                x: x + u32::from(char.width),
269                y: y + u32::from(char.height),
270            },
271        };
272
273        if let Some(character) = char_ids.to_char(char.id) {
274            atlas_character_map.insert(
275                character,
276                AtlasCharacter {
277                    font_char: char,
278                    rect,
279                    placement: None,
280                },
281            );
282        } else {
283            warn!(
284                "Skipping invalid character id {}. Full char definition: {char:?} ",
285                char.id
286            );
287        }
288    }
289
290    atlas_character_map
291}

293/// The textures of a loaded font, with their atlas layouts and outline layers
294/// at the same indices.
295#[derive(Debug, Default)]
296struct FontTextures {
297    /// The font's textures.
298    images: Vec<Handle<Image>>,
299    /// The atlas layout of each texture.
300    atlas_layouts: Vec<Handle<TextureAtlasLayout>>,
301    /// The outline and glyph layers of each texture that holds an outline.
302    outline_layers: Vec<Option<ImageFontOutlineLayers>>,
303}

305/// Loads font images and creates texture atlases.
306///
307/// Grayscale pages are expanded to RGBA, and each page is decoded according to
308/// the font's channel settings. A packed
309/// page becomes one texture per channel its glyphs are in. Textures that hold
310/// an outline are also split into outline and glyph layers.
311async fn load_images_and_textures(
312    bm_font: &bmfont_rs::Font,
313    atlas_character_map: &mut HashMap<char, AtlasCharacter<'_>>,
314    settings: &ImageFontLoaderSettings,
315    load_context: &mut LoadContext<'_>,
316) -> Result<FontTextures, BmFontLoadError> {
317    let channel_settings = ChannelSettings::new(&bm_font.common);
318    let mut textures = FontTextures::default();
319    for (page_no, page) in bm_font.pages.iter().enumerate() {
320        let image_path = load_context
321            .path()
322            .resolve_embed_str(page)
323            .map_err(ImageFontLoadError::from)?;
324
325        let mut image: Image = load_context
326            .load_builder()
327            .with_settings(|settings: &mut ImageLoaderSettings| {
328                settings.expand_grayscale = false;
329            })
330            .load_value(image_path.clone())
331            .await
332            .map_err(ImageFontLoadError::from)?
333            .take();
334
335        image.sampler = settings.image_sampler.clone();
336        let channel_settings = match expand_grayscale_page(&mut image, bm_font.common.alpha_chnl) {
337            Some(alpha) => channel_settings.for_8_bit_page(alpha),
338            None => channel_settings,
339        };
340
341        let size = image.size();
342        if let Some((&character, &AtlasCharacter { rect, .. })) =
343            atlas_character_map.iter().find(|&(_, character)| {
344                usize::from(character.font_char.page) == page_no
345                    && character.rect.max.cmpgt(size).any()
346            })
347        {
348            return Err(BmFontLoadError::CharacterRectOutOfBounds {
349                character,
350                rect,
351                page: page_no,
352                page_size: size,
353            });
354        }
355
356        let page_characters = atlas_character_map
357            .values_mut()
358            .filter(|character| usize::from(character.font_char.page) == page_no);
359        let mut views: Vec<(PageDecoding, Vec<&mut AtlasCharacter<'_>>)> = Vec::new();
360        if let ChannelSettings::NotPacked(decoding) = channel_settings {
361            views.push((decoding, page_characters.collect()));
362        } else {
363            for character in page_characters {
364                let decoding = channel_settings.decoding_for(character.font_char.chnl);
365                match views.iter_mut().find(|&&mut (view, _)| view == decoding) {
366                    Some(&mut (_, ref mut characters)) => characters.push(character),
367                    None => views.push((decoding, vec![character])),
368                }
369            }
370        }
371
372        let view_count = views.len();
373        let mut page_image = Some(image);
374        for (view_no, (decoding, characters)) in views.into_iter().enumerate() {
375            let mut view_image = if view_no + 1 == view_count {
376                page_image.take()
377            } else {
378                page_image.clone()
379            }
380            .unwrap_or_else(|| unreachable!("the page image is taken by the last view only"));
381            decoding.decode_image(&mut view_image);
382
383            let suffix = if channel_settings.is_packed() {
384                decoding.packed_label_suffix()
385            } else {
386                ""
387            };
388            let label = format!("texture_{page_no}{suffix}");
389            let layers = decoding
390                .has_outline()
391                .then(|| outline_layers(&view_image))
392                .flatten()
393                .map(|(outline, glyph)| ImageFontOutlineLayers {
394                    outline: load_context.add_labeled_asset(format!("{label}_outline"), outline),
395                    glyph: load_context.add_labeled_asset(format!("{label}_glyph"), glyph),
396                });
397            textures.outline_layers.push(layers);
398
399            let texture_index = textures.images.len();
400            textures
401                .images
402                .push(load_context.add_labeled_asset(label, view_image));
403
404            let mut atlas_layout = TextureAtlasLayout::new_empty(size);
405            for character in characters {
406                character.placement =
407                    Some((texture_index, atlas_layout.add_texture(character.rect)));
408            }
409            textures.atlas_layouts.push(
410                load_context.add_labeled_asset(format!("layout_{page_no}{suffix}"), atlas_layout),
411            );
412        }
413    }
414
415    Ok(textures)
416}

418/// Constructs the final `ImageFont` asset.
419fn construct_image_font(
420    bm_font: &bmfont_rs::Font,
421    char_ids: CharIds,
422    textures: FontTextures,
423    atlas_character_map: HashMap<char, AtlasCharacter<'_>>,
424    settings: &ImageFontLoaderSettings,
425) -> ImageFont {
426    let mut kernings_by_first: HashMap<u32, Vec<&bmfont_rs::Kerning>> = HashMap::default();
427    for kerning in &bm_font.kernings {
428        kernings_by_first
429            .entry(kerning.first)
430            .or_default()
431            .push(kerning);
432    }
433
434    ImageFont {
435        textures: textures.images,
436        atlas_character_map: atlas_character_map
437            .into_iter()
438            .map(|(char, character)| {
439                let font_char = character.font_char;
440                let kernings = kernings_by_first
441                    .remove(&font_char.id)
442                    .unwrap_or_default()
443                    .into_iter()
444                    .filter_map(|kerning| {
445                        if let Some(character) = char_ids.to_char(kerning.second) {
446                            Some((character, f32::from(kerning.amount)))
447                        } else {
448                            warn!(
449                                "Skipping invalid character id {} in kerning data.",
450                                kerning.second
451                            );
452                            None
453                        }
454                    });
455
456                (
457                    char,
458                    #[expect(
459                        clippy::unwrap_used,
460                        reason = "parse_bmf_data rejects chars on missing pages, so \
461                                  load_images_and_textures places every character"
462                    )]
463                    ImageFontCharacter {
464                        character_index: character.placement.unwrap().1,
465                        page_index: character.placement.unwrap().0,
466                        offsets: Vec2::new(
467                            f32::from(font_char.xoffset),
468                            -f32::from(font_char.yoffset),
469                        ),
470                        x_advance: Some(f32::from(font_char.xadvance)),
471                        kernings: kernings.collect(),
472                    },
473                )
474            })
475            .collect(),
476        atlas_layouts: textures.atlas_layouts,
477        image_sampler: settings.image_sampler.clone(),
478        line_height: Some(u32::from(bm_font.common.line_height)),
479        outline_layers: textures.outline_layers,
480    }
